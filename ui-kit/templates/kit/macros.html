{# tockmox-ui-kit macros: the status language and the source line, one format on every surface. A reading passed to source_line / fold_summary has .state, .source, .reason and .age_text(). #} {# State is never colour alone: a distinct SHAPE, a WORD, and a colour token. #} {% macro state_icon(state) -%} {%- if state == "ok" -%} {%- elif state == "warning" -%} {%- elif state == "critical" -%} {%- else -%} {%- endif -%} {%- endmacro %} {# One state vocabulary across every page (UI standard §2). #} {% macro state_word(state) -%} {{ {"ok": "OK", "warning": "Warning", "critical": "Problem", "unknown": "Unknown"}.get(state, "Unknown") }} {%- endmacro %} {% macro badge(state, word=None, lg=false) -%} {{ state_icon(state) }}{{ word or state_word(state) }} {%- endmacro %} {# Source line: state, source name, reason, and how long since it last answered. One format everywhere. #} {% macro source_line(r) -%} {{ badge(r.state) }}{{ r.source }} {%- if r.reason %}: {{ r.reason }}{% endif -%} · last answered {{ r.age_text() }} {%- endmacro %} {# A healthy group folded to one row: badge, title, source line, chevron. The body is the detail. #} {% macro fold_summary(state, word, title, reading=None, note=None) -%} {%- set why = note if note is not none else (reading.reason if reading else "") -%} {{ badge(state, word) }}{{ title }}{% if reading %}{{ reading.source }}{% if why %}: {{ why }}{% endif %} · last answered {{ reading.age_text() }}{% else %}{% endif %} {%- endmacro %}
